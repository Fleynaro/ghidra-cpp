# Each analyzer owns this source and test registration boundary.
add_library(analyzer_variadic STATIC)
add_library(analyzer_variadic_function_signature_override ALIAS analyzer_variadic)
target_sources(analyzer_variadic PUBLIC FILE_SET cxx_modules TYPE CXX_MODULES FILES src/v.cppm)
target_link_libraries(analyzer_variadic PUBLIC NewGhidra::AnalyzerShared)
target_compile_features(analyzer_variadic PUBLIC cxx_std_23)

if(BUILD_TESTING)
    find_package(GTest CONFIG REQUIRED)
    add_executable(analyzer_variadic_tests)
    add_executable(analyzer_variadic_function_signature_override_tests ALIAS analyzer_variadic_tests)
    target_sources(analyzer_variadic_tests PRIVATE FILE_SET cxx_modules TYPE CXX_MODULES FILES tests/t.cppm)
    target_link_libraries(analyzer_variadic_tests PRIVATE analyzer_variadic GTest::gtest_main)
    target_compile_features(analyzer_variadic_tests PRIVATE cxx_std_23)
    add_test(NAME analyzer_variadic_function_signature_override_tests COMMAND analyzer_variadic_tests)
endif()
